Architecture Frontend VentureLink
Ce document décrit l'architecture complète du frontend de l'application mobile VentureLink, développée avec Flutter/Dart. Il sert de référence principale pour l'équipe frontend et définit l'organisation du code, les patterns utilisés, et les choix techniques.
1. Vue d'ensemble
L'application VentureLink adopte une architecture MVVM (Model-View-ViewModel) permettant une séparation claire des responsabilités et une meilleure testabilité. Le frontend est entièrement développé avec Flutter pour une expérience utilisateur cohérente sur iOS et Android.
1.1 Principes architecturaux
•	Séparation des préoccupations: Chaque composant a une responsabilité unique et bien définie
•	Testabilité: Architecture conçue pour faciliter les tests unitaires et d'intégration
•	Maintenabilité: Organisation claire du code pour une évolution facilitée
•	Réactivité: UI réactive aux changements d'état via des patterns observables
•	Performances: Optimisation des rendus et des appels réseau
1.2 Stack technologique
•	Framework: Flutter 3.10+
•	Langage: Dart 3.0+
•	Gestion d'état: Provider pour la gestion d'état locale et globale
•	Navigation: Flutter Navigation 2.0 avec AutoRoute
•	Réseau: Dio pour les requêtes HTTP
•	Intégrations Firebase: Authentication, Cloud Messaging
•	Stockage local: Shared Preferences et Hive
•	Injection de dépendances: GetIt
•	Internationalisation: Flutter Localizations

11.2 Optimisation des Images
// core/utils/image_utils.dart
class ImageUtils {
  // Compression des images avant upload
  static Future<File> compressImage(File imageFile, {
    int quality = 85,
    int maxWidth = 1080,
    int maxHeight = 1920,
  }) async {
    final dir = await getTemporaryDirectory();
    final targetPath = '${dir.path}/${DateTime.now().millisecondsSinceEpoch}.jpg';
    
    // Compression avec flutter_image_compress
    final result = await FlutterImageCompress.compressAndGetFile(
      imageFile.path,
      targetPath,
      quality: quality,
      minWidth: maxWidth,
      minHeight: maxHeight,
    );
    
    return File(result!.path);
  }
  
  // Chargement des images avec cache
  static CachedNetworkImage cachedImage(
    String url, {
    double? width,
    double? height,
    BoxFit fit = BoxFit.cover,
    Widget Function(BuildContext, String)? placeholder,
    Widget Function(BuildContext, String, dynamic)? errorWidget,
  }) {
    return CachedNetworkImage(
      imageUrl: url,
      width: width,
      height: height,
      fit: fit,
      placeholder: placeholder ?? (context, url) => Container(
        color: Colors.grey[200],
        child: const Center(
          child: CircularProgressIndicator(
            strokeWidth: 2.0,
          ),
        ),
      ),
      errorWidget: errorWidget ?? (context, url, error) => Container(
        color: Colors.grey[200],
        child: const Center(
          child: Icon(Icons.error_outline),
        ),
      ),
    );
  }
}

// Ajouter un gestionnaire de cache
final _cacheStore = MemoryCacheStore(maxSize: 10485760); // 10 MB
final _cacheOptions = CacheOptions(
  store: _cacheStore,
  policy: CachePolicy.request,
  hitCacheOnErrorExcept: [401, 403], // Ne pas utiliser le cache en cas d'erreur d'autorisation
  maxStale: const Duration(days: 1), // Durée maximum du cache
  priority: CachePriority.normal,
  keyBuilder: CacheOptions.defaultCacheKeyBuilder,
);

// Initialiser Dio avec le cache
DioClient(this._dio, this._storageService) {
  _dio.options.baseUrl = AppConfig.apiBaseUrl;
  _dio.options.connectTimeout = const Duration(seconds: 15);
  _dio.options.receiveTimeout = const Duration(seconds: 15);
  _dio.options.responseType = ResponseType.json;
  
  // Ajouter l'intercepteur de cache
  _dio.interceptors.add(DioCacheInterceptor(options: _cacheOptions));
  
  // Autres intercepteurs
  _dio.interceptors.add(_createAuthInterceptor());
  _dio.interceptors.add(_createLogInterceptor());
  _dio.interceptors.add(_createErrorInterceptor());
}

// Méthode pour les requêtes avec cache spécifique
Future<Response> getCached(
  String path, {
  Map<String, dynamic>? queryParameters,
  Duration maxStale = const Duration(hours: 1),
  Duration maxAge = const Duration(minutes: 10),
  bool forceRefresh = false,
}) async {
  final cacheOptions = _cacheOptions.copyWith(
    policy: forceRefresh ? CachePolicy.refresh : CachePolicy.request,
    maxStale: maxStale,
    maxAge: maxAge,
  );
  
  try {
    final response = await _dio.get(
      path,
      queryParameters: queryParameters,
      options: Options(extra: {
        'dio_cache_options': cacheOptions.toExtra(),
      }),
    );
    return response;
  } catch (e) {
    throw _handleError(e);
  }
}

// Méthode pour effacer le cache
Future<void> clearCache() async {
  await _cacheStore.clean();
}

// Méthode pour effacer une entrée spécifique du cache
Future<void> invalidateCacheEntry(String key) async {
  await _cacheStore.delete(key);
}
11.4 Rendering Optimisé
// presentation/common_widgets/optimized_list_view.dart
class OptimizedListView<T> extends StatelessWidget {
  final List<T> items;
  final Widget Function(BuildContext context, T item, int index) itemBuilder;
  final bool isLoading;
  final VoidCallback? onLoadMore;
  final bool hasMoreItems;
  final double itemHeight;
  
  const OptimizedListView({
    Key? key,
    required this.items,
    required this.itemBuilder,
    this.isLoading = false,
    this.onLoadMore,
    this.hasMoreItems = false,
    required this.itemHeight,
  }) : super(key: key);
  
  @override
  Widget build(BuildContext context) {
    return NotificationListener<ScrollNotification>(
      onNotification: (ScrollNotification scrollInfo) {
        if (!isLoading && 
            hasMoreItems && 
            scrollInfo.metrics.pixels >= scrollInfo.metrics.maxScrollExtent * 0.9 &&
            onLoadMore != null) {
          onLoadMore!();
        }
        return false;
      },
      child: ListView.builder(
        // Optimiser les performances avec un contrôleur d'économie de mémoire
        addAutomaticKeepAlives: false,
        addRepaintBoundaries: false,
        itemCount: items.length + (isLoading || hasMoreItems ? 1 : 0),
        // Optimiser le défilement avec une hauteur fixe
        itemExtent: itemHeight,
        // Construire uniquement les éléments visibles et proches
        cacheExtent: MediaQuery.of(context).size.height,
        itemBuilder: (context, index) {
          if (index >= items.length) {
            return const Center(
              child: Padding(
                padding: EdgeInsets.all(16.0),
                child: CircularProgressIndicator(),
              ),
            );
          }
          
          // Utiliser RepaintBoundary pour isoler le rendering
          return RepaintBoundary(
            child: itemBuilder(context, items[index], index),
          );
        },
      ),
    );
  }
}

// services/security_service.dart
class SecurityService {
  final FlutterSecureStorage _secureStorage;
  
  SecurityService(this._secureStorage);
  
  // Stockage de données sensibles
  Future<void> secureWrite(String key, String value) async {
    await _secureStorage.write(key: key, value: value);
  }
  
  Future<String?> secureRead(String key) async {
    return await _secureStorage.read(key: key);
  }
  
  Future<void> secureDelete(String key) async {
    await _secureStorage.delete(key: key);
  }
  
  // Chiffrement des données
  Future<String> encrypt(String plainText, String key) async {
    final encrypter = _createEncrypter(key);
    final iv = IV.fromLength(16);
    final encrypted = encrypter.encrypt(plainText, iv: iv);
    return encrypted.base64;
  }
  
  Future<String> decrypt(String encryptedText, String key) async {
    final encrypter = _createEncrypter(key);
    final iv = IV.fromLength(16);
    final encrypted = Encrypted.fromBase64(encryptedText);
    return encrypter.decrypt(encrypted, iv: iv);
  }
  
  Encrypter _createEncrypter(String key) {
    final keyBytes = md5.convert(utf8.encode(key)).bytes;
    final encryptionKey = Key(Uint8List.fromList(keyBytes));
    return Encrypter(AES(encryptionKey));
  }
  
  // Génération de clé sécurisée
  Future<String> generateSecureKey() async {
    final random = Random.secure();
    final values = List<int>.generate(32, (i) => random.nextInt(256));
    return base64Url.encode(values);
  }
}
12.2 Protection contre le jailbreak/root
// services/security_service.dart (ajout)
Future<bool> isDeviceSecure() async {
  if (Platform.isAndroid) {
    return !await _isRooted();
  } else if (Platform.isIOS) {
    return !await _isJailbroken();
  }
  return true;
}

Future<bool> _isRooted() async {
  try {
    // Vérifier les chemins de fichiers communs aux appareils rootés
    const paths = [
      '/system/app/Superuser.apk',
      '/sbin/su',
      '/system/bin/su',
      '/system/xbin/su',
      '/data/local/xbin/su',
      '/data/local/bin/su',
      '/system/sd/xbin/su',
      '/system/bin/failsafe/su',
      '/data/local/su'
    ];
    
    for (final path in paths) {
      if (await File(path).exists()) {
        return true;
      }
    }
    
    // Vérifier les applications de root connues
    final pm = await channel.invokeMethod('checkPackageManager');
    final packages = [
      'com.noshufou.android.su',
      'com.thirdparty.superuser',
      'eu.chainfire.supersu',
      'com.koushikdutta.superuser',
      'com.zachspong.temprootremovejb',
      'com.ramdroid.appquarantine'
    ];
    
    for (final pkg in packages) {
      if (pm.contains(pkg)) {
        return true;
      }
    }
    
    return false;
  } catch (e) {
    // En cas d'erreur, considérer l'appareil comme potentiellement compromis
    return true;
  }
}

Future<bool> _isJailbroken() async {
  try {
    // Vérifier les chemins de fichiers communs aux appareils jailbreakés
    const paths = [
      '/Applications/Cydia.app',
      '/Library/MobileSubstrate/MobileSubstrate.dylib',
      '/bin/bash',
      '/usr/sbin/sshd',
      '/etc/apt',
      '/private/var/lib/apt/'
    ];
    
    for (final path in paths) {
      if (await File(path).exists()) {
        return true;
      }
    }
    
    // Vérifier si l'application peut écrire en dehors de son sandbox
    final testFile = '/private/jailbreak.txt';
    try {
      await File(testFile).writeAsString('test');
      await File(testFile).delete();
      return true;
    } catch (e) {
      // Normal, ne pas pouvoir écrire ici
    }
    
    return false;
  } catch (e) {
    // En cas d'erreur, considérer l'appareil comme potentiellement compromis
    return true;
  }
}

class CertificatePinning {
  // Configuration du pinning de certificat avec dio_pinning
  static void setupPinning(Dio dio) {
    // SHA-256 des certificats de l'API VentureLink
    final fingerprints = [
      'sha256/AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=', // Remplacer par l'empreinte réelle
      'sha256/BBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBB=', // Certificat secondaire/backup
    ];
    
    // Ajouter l'intercepteur de pinning
    dio.httpClientAdapter = PinningAdapter(
      allowedSHAFingerprints: fingerprints,
      // Domaines à vérifier
      trustedDomains: ['api.venturelink.com', 'api-dev.venturelink.com'],
    );
  }
}

13.1 Configuration de l'environnement
// config/environment.dart
enum Environment {
  development,
  staging,
  production,
}

class AppConfig {
  static late Environment environment;
  static late String apiBaseUrl;
  static late String apiKey;
  static late bool enableAnalytics;
  static late bool enableCrashlytics;
  
  static void configure(Environment env) {
    environment = env;
    
    switch (environment) {
      case Environment.development:
        apiBaseUrl = 'https://api-dev.venturelink.com/api/v1';
        apiKey = 'dev-api-key';
        enableAnalytics = false;
        enableCrashlytics = false;
        break;
      case Environment.staging:
        apiBaseUrl = 'https://api-staging.venturelink.com/api/v1';
        apiKey = 'staging-api-key';
        enableAnalytics = true;
        enableCrashlytics = true;
        break;
      case Environment.production:
        apiBaseUrl = 'https://api.venturelink.com/api/v1';
        apiKey = 'prod-api-key';
        enableAnalytics = true;
        enableCrashlytics = true;
        break;
    }
  }
  
  static bool get isProduction => environment == Environment.production;
  static bool get isStaging => environment == Environment.staging;
  static bool get isDevelopment => environment == Environment.development;
}

// Entrées principales pour chaque environnement
// main_dev.dart
void main() {
  AppConfig.configure(Environment.development);

}

void main() {

      final result = await _getProjectUseCase(projectId);
      _project = result;
      _loadQuestions(projectId);
      _error = null;

    } finally {
      _setLoading(false);

      final result = await _getQuestionsUseCase(projectId);
      _questions = result;
      notifyListeners();
    } catch (e) {
      // Gérer l'erreur (mais ne pas bloquer l'affichage du projet)
      _questions = [];
      notifyListeners();
    }
  }
  
  Future<void> askQuestion(String questionText) async {
    try {
      final newQuestion = await _askQuestionUseCase(
        project.id,
        questionText,
      );
      
      _questions = [..._questions, newQuestion];
      notifyListeners();
    } catch (e) {
      // Gérer l'erreur
      rethrow;
    }
  }
  
  Future<void> expressInterest() async {
    try {
      await _expressInterestUseCase(project.id);
      // Recharger le projet pour mettre à jour le statut
      await loadProject(project.id);
    } catch (e) {
      // Gérer l'erreur
      rethrow;
    }
  }
  
  void _setLoading(bool value) {
    _isLoading = value;
    notifyListeners();
  }

// data/repositories/project_repository.dart
class ProjectRepository implements IProjectRepository {
  final ProjectRemoteDataSource _remoteDataSource;
  final ProjectLocalDataSource _localDataSource;
  final NetworkInfo _networkInfo;
  
  ProjectRepository(
    this._remoteDataSource,
    this._localDataSource,
    this._networkInfo,
  );
  
  @override
  Future<Project> getProject(String id) async {
    if (await _networkInfo.isConnected) {
      try {
        final remoteProject = await _remoteDataSource.getProject(id);
        await _localDataSource.cacheProject(remoteProject);
        return remoteProject.toDomain();
      } catch (e) {
        // Essayer de récupérer les données en cache si disponibles
        if (await _localDataSource.hasProjectCached(id)) {
          final localProject = await _localDataSource.getProject(id);
          return localProject.toDomain();
        }
        rethrow;
      }
    } else {
      // Pas de connexion, utiliser les données en cache
      if (await _localDataSource.hasProjectCached(id)) {
        final localProject = await _localDataSource.getProject(id);
        return localProject.toDomain();
      }
      throw NoInternetConnectionException();
    }
  }
  
  // Autres implémentations...
}

class AuthProvider extends ChangeNotifier {
  final AuthService _authService;
  final StorageService _storageService;
  
  bool _isAuthenticated = false;
  bool _isLoading = false;
  Exception? _error;
  
  // Getters
  bool get isAuthenticated => _isAuthenticated;
  bool get isLoading => _isLoading;
  Exception? get error => _error;
  
  AuthProvider({
    required AuthService authService,
    required StorageService storageService,
  }) : _authService = authService,
       _storageService = storageService {
    // Initialiser l'état d'authentification
    _checkAuthStatus();
  }
  
  Future<void> _checkAuthStatus() async {
    final token = await _storageService.getToken();
    
    if (token != null) {
      _isAuthenticated = true;
      notifyListeners();
    }
  }
  
  Future<void> login(String email, String password) async {
    _setLoading(true);
    
    try {
      final result = await _authService.loginWithEmailPassword(email, password);
      
      // Stocker le token
      await _storageService.saveToken(result.token);
      await _storageService.saveRefreshToken(result.refreshToken);
      
      _isAuthenticated = true;
      _error = null;
    } catch (e) {
      _isAuthenticated = false;
      _error = e is Exception ? e : Exception('Une erreur est survenue');

      _setLoading(false);
    }

// services/firebase/auth_service.dart
class AuthService {
  final FirebaseAuth _firebaseAuth = FirebaseAuth.instance;
  final GoogleSignIn _googleSignIn = GoogleSignIn();
  final ApiService _apiService;
  
  AuthService(this._apiService);
  
  // Authentification par email/password via Firebase
  Future<AuthResult> loginWithEmailPassword(String email, String password) async {
    try {
      // Authentifier avec Firebase
      final userCredential = await _firebaseAuth.signInWithEmailAndPassword(
        email: email,
        password: password,
      );
      
      // Obtenir le token Firebase
      final firebaseToken = await userCredential.user!.getIdToken();
      
      // Authentifier avec notre backend
      final response = await _apiService.loginWithFirebase(
        provider: 'EMAIL',
        token: firebaseToken,
      );
      
      return AuthResult(
        userId: response.id,
        token: response.token,
        refreshToken: response.refreshToken,
        isNewUser: false,
      );
    } catch (e) {
      // Gérer les erreurs spécifiques
      if (e is FirebaseAuthException) {
        if (e.code == 'user-not-found' || e.code == 'wrong-password') {
          throw InvalidCredentialsException();
        }
      }
      rethrow;
    }
  }
  
  // Enregistrement avec email/password
  Future<AuthResult> register({
    required String email,
    required String password,
    required String firstName,
    required String lastName,
    required UserType userType,
  }) async {
    try {
      // Créer l'utilisateur dans Firebase
      final userCredential = await _firebaseAuth.createUserWithEmailAndPassword(
        email: email,
        password: password,
      );
      
      // Mettre à jour le profil Firebase
      await userCredential.user!.updateDisplayName('$firstName $lastName');
      
      // Obtenir le token Firebase
      final firebaseToken = await userCredential.user!.getIdToken();
      
      // Enregistrer avec notre backend
      final response = await _apiService.registerWithFirebase(
        provider: 'EMAIL',
        token: firebaseToken,
        firstName: firstName,
        lastName: lastName,
        userType: userType.toString().split('.').last,
      );
      
      return AuthResult(
        userId: response.id,
        token: response.token,
        refreshToken: response.refreshToken,
        isNewUser: true,
      );
    } catch (e) {
      // Gérer les erreurs spécifiques
      if (e is FirebaseAuthException) {
        if (e.code == 'email-already-in-use') {
          throw EmailAlreadyInUseException();
        }
      }
      rethrow;
    }
  }
  
  // Authentification Google
  Future<AuthResult> loginWithGoogle() async {
    try {
      // Déclencher le flow d'authentification Google
      final GoogleSignInAccount? googleUser = await _googleSignIn.signIn();
      
      if (googleUser == null) {
        throw CancelledException();
      }
      
      // Obtenir les détails d'authentification
      final googleAuth = await googleUser.authentication;
      
      // Créer un credential Firebase
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );
      
      // S'authentifier avec Firebase
      final userCredential = await _firebaseAuth.signInWithCredential(credential);
      
      // Obtenir le token Firebase
      final firebaseToken = await userCredential.user!.getIdToken();
      
      // Authentifier avec notre backend
      final response = await _apiService.loginWithFirebase(
        provider: 'GOOGLE',
        token: firebaseToken,
      );
      
      return AuthResult(
        userId: response.id,
        token: response.token,
        refreshToken: response.refreshToken,
        isNewUser: response.isNewUser,
      );
    } catch (e) {
      if (e is CancelledException) {
        rethrow;
      }
      throw AuthenticationException('Échec de l\'authentification Google');
    }
  }
  
  // Authentification LinkedIn
  Future<AuthResult> loginWithLinkedIn() async {
    try {
      // Utiliser le package flutter_web_auth pour OAuth LinkedIn
      final result = await FlutterWebAuth.authenticate(
        url: '${AppConfig.apiBaseUrl}/api/v1/auth/linkedin/authorize',
        callbackUrlScheme: 'venturelink',
      );
      
      // Extraire le token
      final token = Uri.parse(result).queryParameters['token'];
      
      if (token == null) {
        throw AuthenticationException('Token LinkedIn manquant');
      }
      
      // Échanger le token avec notre backend
      final response = await _apiService.loginWithLinkedIn(token: token);
      
      return AuthResult(
        userId: response.id,
        token: response.token,
        refreshToken: response.refreshToken,
        isNewUser: response.isNewUser,
      );
    } catch (e) {
      throw AuthenticationException('Échec de l\'authentification LinkedIn');
    }
  }
  
  // Déconnexion
  Future<void> logout() async {
    await _firebaseAuth.signOut();
    await _googleSignIn.signOut();
    await _apiService.logout();
  }

class StorageService {

  // Tokens (stockés en sécurité)

  }

    await _secureStorage.write(key: _tokenKey, value: token);
  }

  }
  
  Future<String?> getRefreshToken() async {

  }
  
  Future<void> saveRefreshToken(String token) async {
    await _secureStorage.write(key: _refreshTokenKey, value: token);
  }
  
  Future<void> deleteRefreshToken() async {
    await _secureStorage.delete(key: _refreshTokenKey);
  }
  
  // Informations utilisateur
  Future<String?> getUserId() async {
    return _prefs.getString(_userIdKey);
  }
  
  Future<void> saveUserId(String userId) async {
    await _prefs.setString(_userIdKey, userId);
  }
  
  Future<void> deleteUserId() async {
    await _prefs.remove(_userIdKey);
  }
  
  Future<UserType?> getUserType() async {
    final userTypeStr = _prefs.getString(_userTypeKey);
    if (userTypeStr == null) return null;
    return UserTypeExtension.fromString(userTypeStr);
  }
  
  Future<void> saveUserType(UserType userType) async {
    await _prefs.setString(_userTypeKey, userType.toString().split('.').last);
  }
  
  // FCM Token
  Future<String?> getFCMToken() async {
    return _prefs.getString(_fcmTokenKey);
  }
  
  Future<void> saveFCMToken(String token) async {
    await _prefs.setString(_fcmTokenKey, token);
  }
  
  // Préférences
  Future<ThemeMode> getThemeMode() async {
    final themeStr = _prefs.getString(_themeKey);
    if (themeStr == null) return ThemeMode.system;
    
    switch (themeStr) {
      case 'light':
        return ThemeMode.light;
      case 'dark':
        return ThemeMode.dark;
      default:
        return ThemeMode.system;
    }
  }
  
  Future<void> saveThemeMode(ThemeMode themeMode) async {
    String themeStr;
    
    switch (themeMode) {
      case ThemeMode.light:
        themeStr = 'light';
        break;
      case ThemeMode.dark:
        themeStr = 'dark';
        break;
      case ThemeMode.system:
        themeStr = 'system';
        break;
    }
    
    await _prefs.setString(_themeKey, themeStr);
  }
  
  Future<Locale?> getLocale() async {
    final localeStr = _prefs.getString(_localeKey);
    if (localeStr == null) return null;
    
    final parts = localeStr.split('_');
    if (parts.length == 1) {
      return Locale(parts[0]);
    } else if (parts.length == 2) {
      return Locale(parts[0], parts[1]);
    }
    
    return null;
  }
  
  Future<void> saveLocale(Locale locale) async {
    final localeStr = locale.countryCode == null
        ? locale.languageCode
        : '${locale.languageCode}_${locale.countryCode}';
    
    await _prefs.setString(_localeKey, localeStr);
  }
  
  // Nettoyage complet (déconnexion)
  Future<void> clearAllData() async {
    await _secureStorage.deleteAll();
    await _prefs.clear();
  }
}

class DioClient {
  final Dio _dio;
  final StorageService _storageService;
  
  DioClient(this._dio, this._storageService) {
    _dio.options.baseUrl = AppConfig.apiBaseUrl;
    _dio.options.connectTimeout = const Duration(seconds: 15);
    _dio.options.receiveTimeout = const Duration(seconds: 15);
    _dio.options.responseType = ResponseType.json;
    
    // Ajouter les intercepteurs
    _dio.interceptors.add(_createAuthInterceptor());
    _dio.interceptors.add(_createLogInterceptor());
    _dio.interceptors.add(_createErrorInterceptor());
  }
  
  // Méthodes HTTP
  Future<Response> get(
    String path, {
    Map<String, dynamic>? queryParameters,
    Options? options,
  }) async {
    try {
      final response = await _dio.get(
        path,
        queryParameters: queryParameters,
        options: options,
      );
      return response;
    } catch (e) {
      throw _handleError(e);
    }
  }
  
  Future<Response> post(
    String path, {
    dynamic data,
    Map<String, dynamic>? queryParameters,
    Options? options,
  }) async {
    try {
      final response = await _dio.post(
        path,
        data: data,
        queryParameters: queryParameters,
        options: options,
      );
      return response;
    } catch (e) {
      throw _handleError(e);
    }
  }
  
  Future<Response> put(
    String path, {
    dynamic data,
    Map<String, dynamic>? queryParameters,
    Options? options,
  }) async {
    try {
      final response = await _dio.put(
        path,
        data: data,
        queryParameters: queryParameters,
        options: options,
      );
      return response;
    } catch (e) {
      throw _handleError(e);
    }
  }
  
  Future<Response> delete(
    String path, {
    dynamic data,
    Map<String, dynamic>? queryParameters,
    Options? options,
  }) async {
    try {
      final response = await _dio.delete(
        path,
        data: data,
        queryParameters: queryParameters,
        options: options,
      );
      return response;
    } catch (e) {
      throw _handleError(e);
    }
  }
  
  // Upload de fichiers
  Future<Response> uploadFile(
    String path,
    File file, {
    String paramName = 'file',
    Map<String, dynamic>? fields,
    ProgressCallback? onSendProgress,
  }) async {
    try {
      final fileName = file.path.split('/').last;
      final formData = FormData.fromMap({
        paramName: await MultipartFile.fromFile(
          file.path,
          filename: fileName,
        ),
        if (fields != null) ...fields,
      });
      
      final response = await _dio.post(
        path,
        data: formData,
        onSendProgress: onSendProgress,
      );
      return response;
    } catch (e) {
      throw _handleError(e);
    }
  }
  
  // Intercepteurs
  Interceptor _createAuthInterceptor() {
    return InterceptorsWrapper(
      onRequest: (options, handler) async {
        // Ajouter le token d'authentification si disponible
        final token = await _storageService.getToken();
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        
        // Ajouter l'en-tête d'acceptation de langue
        final locale = await _storageService.getLocale();
        if (locale != null) {
          options.headers['Accept-Language'] = locale.languageCode;
        } else {
          options.headers['Accept-Language'] = 'fr';
        }
        
        return handler.next(options);
      },
      onError: (error, handler) async {
        // Gérer l'expiration du token
        if (error.response?.statusCode == 401) {
          // Essayer de rafraîchir le token
          final refreshToken = await _storageService.getRefreshToken();
          
          if (refreshToken != null) {
            try {
              // Appeler l'API de rafraîchissement
              final response = await _dio.post(
                '/api/v1/auth/refresh-token',
                data: {'refresh_token': refreshToken},
                options: Options(headers: {}),
              );
              
              // Stocker le nouveau token
              final newToken = response.data['token'];
              final newRefreshToken = response.data['refresh_token'];
              
              await _storageService.saveToken(newToken);
              await _storageService.saveRefreshToken(newRefreshToken);
              
              // Répéter la requête originale avec le nouveau token
              error.requestOptions.headers['Authorization'] = 'Bearer $newToken';
              
              final opts = Options(
                method: error.requestOptions.method,
                headers: error.requestOptions.headers,
              );
              
              final cloneReq = await _dio.request(
                error.requestOptions.path,
                options: opts,
                data: error.requestOptions.data,
                queryParameters: error.requestOptions.queryParameters,
              );
              
              return handler.resolve(cloneReq);
            } catch (e) {
              // Échec du rafraîchissement, déconnexion
              await _storageService.deleteToken();
              await _storageService.deleteRefreshToken();
              
              return handler.next(error);
            }
          }
        }
        
        return handler.next(error);
      },
    );
  }
  
  Interceptor _createLogInterceptor() {
    return LogInterceptor(
      requestBody: true,
      responseBody: true,
      logPrint: (log) {
        // Utiliser un logger personnalisé
        if (AppConfig.environment == Environment.development) {
          debugPrint('DioClient: $log');
        }
      },
    );
  }
  
  Interceptor _createErrorInterceptor() {
    return InterceptorsWrapper(
      onError: (error, handler) {
        // Personnaliser l'erreur avant de la propager
        return handler.next(error);
      },
    );
  }
  
  // Gestion des erreurs
  Exception _handleError(dynamic error) {
    if (error is DioException) {
      // Erreurs Dio
      switch (error.type) {
        case DioExceptionType.connectionTimeout:
        case DioExceptionType.sendTimeout:
        case DioExceptionType.receiveTimeout:
          return TimeoutException('La connexion a expiré');
          
        case DioExceptionType.badCertificate:
        case DioExceptionType.connectionError:
          return NetworkException('Erreur de connexion');
          
        case DioExceptionType.badResponse:
          return _handleApiError(error.response);
          
        case DioExceptionType.cancel:
          return CancelledException('Requête annulée');
          
        default:
          return UnknownException('Une erreur est survenue');
      }
    }
    
    // Autres erreurs
    return UnknownException('Une erreur est survenue');
  }
  
  Exception _handleApiError(Response? response) {
    if (response == null) {
      return ServerException('Erreur serveur');
    }
    
    final statusCode = response.statusCode;
    
    try {
      // Essayer de parser l'erreur depuis la réponse
      final data = response.data;
      final errorResponse = ErrorResponse.fromJson(data);
      
      switch (statusCode) {
        case 400:
          return ValidationException(errorResponse.message, errorResponse.details);
        case 401:
          return UnauthorizedException(errorResponse.message);
        case 403:
          return ForbiddenException(errorResponse.message);
        case 404:
          return NotFoundException(errorResponse.message);
        case 409:
          return ConflictException(errorResponse.message);
        case 422:
          return ValidationException(errorResponse.message, errorResponse.details);
        case 429:
          return RateLimitExceededException(errorResponse.message);
        case 500:
        default:
          return ServerException(errorResponse.message);
      }
    } catch (e) {
      // Si impossible de parser l'erreur, utiliser le code HTTP
      switch (statusCode) {
        case 400:
          return ValidationException('Requête invalide');
        case 401:
          return UnauthorizedException('Non autorisé');
        case 403:
          return ForbiddenException('Accès refusé');
        case 404:
          return NotFoundException('Ressource introuvable');
        case 409:
          return ConflictException('Conflit');
        case 422:
          return ValidationException('Données invalides');
        case 429:
          return RateLimitExceededException('Trop de requêtes');
        case 500:
        default:
          return ServerException('Erreur serveur');
      }
    }
  }
}

class ConnectivityInterceptor extends Interceptor {
  final NetworkInfo _networkInfo;
  
  ConnectivityInterceptor(this._networkInfo);
  
  @override
  void onRequest(RequestOptions options, RequestInterceptorHandler handler) async {
    if (await _networkInfo.isConnected) {
      // Continuer la requête
      return handler.next(options);
    } else {
      // Annuler la requête avec une erreur
      return handler.reject(
        DioException(
          requestOptions: options,
          error: NoInternetConnectionException(),
          type: DioExceptionType.connectionError,
        ),
      );
    }
  }

// core/network/interceptors/retry_interceptor.dart
class RetryInterceptor extends Interceptor {
  final Dio dio;
  final int retries;
  final Duration retryDelay;
  
  RetryInterceptor(this.dio, {this.retries = 3, this.retryDelay = const Duration(seconds: 1)});
  
  @override
  void onError(DioException err, ErrorInterceptorHandler handler) async {
    // Vérifier si l'erreur est éligible pour une nouvelle tentative
    if (_shouldRetry(err)) {
      final options = err.requestOptions;
      
      // Récupérer les tentatives déjà effectuées
      final extraInfo = options.extra;
      final retryCount = extraInfo['retryCount'] ?? 0;
      
      // Vérifier si on a atteint le nombre maximal de tentatives
      if (retryCount < retries) {
        // Incrémenter le compteur
        options.extra['retryCount'] = retryCount + 1;
        
        // Attendre avant de réessayer
        await Future.delayed(retryDelay * (retryCount + 1));
        
        try {
          // Réessayer la requête
          final response = await dio.fetch(options);
          return handler.resolve(response);
        } catch (e) {
          // Si cela échoue à nouveau, continuer avec l'erreur
          if (e is DioException) {
            return handler.next(e);
          }
          return handler.next(err);
        }
      }
    }
    
    // Continuer avec l'erreur
    return handler.next(err);
  }
  
  bool _shouldRetry(DioException error) {
    // Réessayer en cas d'erreurs réseau ou de timeout
    return error.type == DioExceptionType.connectionTimeout ||
        error.type == DioExceptionType.receiveTimeout ||
        error.type == DioExceptionType.sendTimeout ||
        error.type == DioExceptionType.connectionError ||
        (error.type == DioExceptionType.badResponse && 
         error.response?.statusCode == 500); // Erreur serveur
  }
}

// core/error/error_handler.dart
class ErrorHandler {
  static String getErrorMessage(Exception exception) {
    if (exception is AppException) {
      return exception.message;
    }
    
    return 'Une erreur est survenue';
  }
  
  static bool isNetworkError(Exception exception) {
    return exception is NetworkException ||
           exception is TimeoutException ||
           exception is NoInternetConnectionException;
  }
  
  static bool isAuthenticationError(Exception exception) {
    return exception is UnauthorizedException ||
           exception is AuthenticationException;
  }
  
  static bool isValidationError(Exception exception) {
    return exception is ValidationException;
  }
  
  static List<ValidationError>? getValidationErrors(Exception exception) {
    if (exception is ValidationException) {
      return exception.errors;
    }
    
    return null;
  }
  
  static Widget buildErrorWidget(
    BuildContext context,
    Exception exception, {
    VoidCallback? onRetry,
  }) {
    final errorMessage = getErrorMessage(exception);
    
    if (isNetworkError(exception)) {
      return NetworkErrorView(
        message: errorMessage,
        onRetry: onRetry,
      );
    }
    
    if (isAuthenticationError(exception)) {
      return AuthErrorView(
        message: errorMessage,
        onLogin: () => context.router.replace(const LoginRoute()),
      );
    }
    
    return GeneralErrorView(
      message: errorMessage,
      onRetry: onRetry,
    );
  }
}

class LocalizationConfig {
  static const supportedLocales = [
    Locale('fr'), // Français (par défaut)
    Locale('en'), // Anglais
  ];
  
  static const defaultLocale = Locale('fr');
  
  static const localizationsDelegates = [
    AppLocalizations.delegate,
    GlobalMaterialLocalizations.delegate,
    GlobalWidgetsLocalizations.delegate,
    GlobalCupertinoLocalizations.delegate,
  ];
  
  static bool isSupported(Locale locale) {
    return supportedLocales.contains(locale);
  }
}

// localization/locale_provider.dart
class LocaleProvider extends ChangeNotifier {
  final StorageService _storageService;
  
  Locale? _locale;
  
  LocaleProvider(this._storageService) {
    _loadSavedLocale();
  }
  
  Locale get locale => _locale ?? LocalizationConfig.defaultLocale;
  
  Future<void> _loadSavedLocale() async {
    final savedLocale = await _storageService.getLocale();
    if (savedLocale != null && LocalizationConfig.isSupported(savedLocale)) {
      _locale = savedLocale;
      notifyListeners();
    }
  }
  
  Future<void> setLocale(Locale locale) async {
    if (!LocalizationConfig.isSupported(locale)) return;
    
    _locale = locale;
    await _storageService.saveLocale(locale);
    notifyListeners();
  }
  
  Future<void> resetLocale() async {
    _locale = null;
    await _storageService.deleteLocale();
    notifyListeners();
  }
}
